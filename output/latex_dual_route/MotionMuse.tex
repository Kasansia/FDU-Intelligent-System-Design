% 04
\begin{frame}[t]{路线 A：ArUco 定位原理与升级}
\begin{columns}[T,onlytextwidth]
\begin{column}{.48\textwidth}
\lead{以可识别标记建立板面坐标}
\begin{points}
\item 手部模型输出双手关键点。
\item 检测标记角点，计算透视变换，将手部位置映射到板面。
\item 多标记角点联合定位，支持自由贴标和部分遮挡。
\item 定位质量不足时暂停新发声。
\end{points}
\end{column}
\begin{column}{.48\textwidth}
\figureplaceholder{04-aruco.png}{4.65cm}{标记与坐标映射}{标出可见角点、板面坐标及手指投影位置}
\end{column}
\end{columns}
\source{实现依据：aruco\_tracking.py、pipeline.py 及项目 README。}
\end{frame}

% 08
\begin{frame}[t]{路线 B：现有无标记定位原型}
\begin{columns}[T,onlytextwidth]
\begin{column}{.49\textwidth}
\begin{points}
\item 冻结画面，人工点击板子四角。
\item 逐帧检测矩形轮廓，结合外观、长宽比和位置连续性选择目标。
\item 从最近可靠检测帧进行短时光流预测，默认窗口约 0.28 秒。
\item 超时未能重新检测时，不再输出定位。
\end{points}
\small\color{FudanBlue}当前采用传统视觉定位。\par 板角训练模型尚未接入。
\end{column}
\begin{column}{.47\textwidth}
\figureplaceholder{08-markerless.png}{4.15cm}{无标记定位示例}{四角初始化、轮廓检测与遮挡恢复}
\end{column}
\end{columns}
\source{3 项合成定位测试通过。完整演示仍需修复启动路径、帧同步与几何一致性问题。}
\end{frame}

\small 视频采集、抽帧、CVAT 标注说明及 YOLO 数据集转换。采集覆盖慢速、演奏、快速、遮挡与无板场景。
